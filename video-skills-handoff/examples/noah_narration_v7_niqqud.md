קָרְאוּ לוֹ נֹחַ, ״זֶה יְנַחֲמֵנוּ״, וְעַל כְּתֵפָיו הַקְּטַנּוֹת הִנִּיחוּ עוֹלָם שָׁלֵם שֶׁל צִפִּיָּה — עַד שֶׁיּוֹם אֶחָד בִּקֵּשׁ הוּא עַצְמוֹ מְעַט מְנוּחָה.

שֵׁשׁ מֵאוֹת שָׁנָה הִכִּיר אֶת עוֹלָמוֹ בְּעַל־פֶּה: אֶת שְׁמוֹת הַשְּׁכֵנִים, אֶת רֵיחַ הַלֶּחֶם עִם עֲלוֹת הַשַּׁחַר, וְאֶת הַיְּלָדִים שֶׁגָּדְלוּ מוּל הַתֵּבָה.

נֹחַ בּוֹנֶה תֵּבָה כִּי שָׁמַע ״קֵץ כָּל בָּשָׂר״. מֵאָה וְעֶשְׂרִים שָׁנָה הִתְרִיעַ בָּהּ, לֹא בִּצְעָקָה אֶלָּא בְּקוֹרוֹת, וַאֲפִלּוּ אֶת הַחֹדֶשׁ כָּתַב לָהֶם, בְּפֶחָם, עַל קֶרֶשׁ: אוֹקְטוֹבֶּר. בַּסְּתָו.

וְהֵם צָחֲקוּ. הִתְחִילוּ מֵהַשֵּׁם עַצְמוֹ, שְׁתֵּי אוֹתִיּוֹת בִּלְבַד, כַּמָּה קָטָן; הִמְשִׁיכוּ בְּ״זָקֵן, תֵּבָה זוֹ לָמָּה?״ וּבְ״תִּרְגַּע, נֹחַ״. קָרְאוּ לוֹ צַדִּיק בִּסְוֶדֶר, מִפְּנֵי שֶׁקַּר לוֹ וְהִתְעַטֵּף, וְאָמְרוּ: ״זֶה הֲכִי נֹחַ — לִבְנוֹת תֵּבָה בְּאֶמְצַע הַיַּבָּשָׁה, וּלְדַבֵּר עַל מַבּוּל שֶׁאִישׁ אֵינוֹ רוֹאֶה.״

וּבְקָצֶה הַגְּבוּל עָמְדוּ צוֹפוֹת, וְהֵן רָאוּ אֶת הַמַּיִם מִתְכַּנְּסִים מֵעֵבֶר לַגִּבְעָה, וְאָמְרוּ, וְחָזְרוּ וְאָמְרוּ, וְהִצְבִּיעוּ, וּבִתְמוּרָה נֶאֱמַר לָהֶן ״אַל תַּגְזִימוּ״, וְיֵשׁ שֶׁבִּקְשׁוּ אֲפִלּוּ לְהַעֲנִישׁ אוֹתָן עַל הַחֻצְפָּה.

וְאָז הִגִּיעַ ״לְשִׁבְעַת הַיָּמִים״ — יוֹם עֲצֶרֶת, וְהַכֹּל נֶעֱצָר, וְהַבֹּקֶר יָפֶה: בַּשָּׂדֶה עוֹד רָקְדוּ עַד אוֹר, נוֹבָה יִקְרְאוּ לָזֶה אַחַר כָּךְ, בַּקִּבּוּצִים הַקָּפֶה עֲדַיִן חַם, וּבִשְׂדֵרוֹת לָבְשׁוּ הַיְּלָדִים בִּגְדֵי חַג, עַד שֶׁ״נִּבְקְעוּ מַעְיְנֹת תְּהוֹם״, וְהַצְּעָקוֹת נִהְיוּ רַעַשׁ, וְהָרַעַשׁ — דְּמָמָה. תֵּבָה אַחַת הָיְתָה לְנֹחַ; לְכָל הָאֲחֵרִים הָיוּ מִגּוּנִּיּוֹת, וְאַף אַחַת מֵהֶן לֹא הָיְתָה תֵּבָה. ״וַיִּסְגֹּר ה׳ בַּעֲדוֹ״, וְהוּא, הַמַּצִּיל, לֹא יָכוֹל הָיָה לִפְתֹּחַ.

שְׁנֵים עָשָׂר חֹדֶשׁ לֹא טָעַם טַעַם שֵׁנָה, וְלֹא שָׂם לֵב, כִּי כָּל עוֹד הַיָּדַיִם עֲסוּקוֹת הַלֵּב אֵינוֹ שׁוֹאֵל; הוּא צוֹעֵד אֶל קָצֶה בְּלִי לָדַעַת, כִּי הָעֲשִׂיָּה אֵינָהּ רִפּוּי אֶלָּא גָּדֵר — עַד שֶׁהָעֲבוֹדָה תִּגָּמֵר וְהַמְּצִיאוּת תִּפְתַּח דֶּלֶת קְטַנָּה אֶל הַלֵּב, וְאָז יָבִין מָה עָבַר עָלָיו. ״וַיִּשָּׁאֶר אַךְ נֹחַ״: הַנִּצּוֹל הָרִאשׁוֹן, הַפּוֹסְט־טְרַאוּמָטִי הָרִאשׁוֹן, וְהַגָּדוֹל.

בֵּינְתַיִם הַמַּיִם סְבִיבוֹ גַּלִּים גַּלִּים, וְגַם בַּלֵּב, שֶׁכְּבָר מְכֻסֶּה אַף שֶׁהַכֹּל עֲדַיִן לֹא שׁוֹקֵעַ, וְיֵשׁ פַּחַד אֶחָד גָּדוֹל מִכָּל גַּל: שֶׁמֵּי הַתְּהוֹם יִבְלְעוּ אוֹתוֹ, וְשֶׁכְּשֶׁיֵּרְדוּ לֹא יִהְיֶה לוֹ אֵיפֹה לְהַנִּיחַ אֶת כַּף הָרֶגֶל. וְאֵיךְ יִישַׁן, כְּשֶׁדֶּרֶךְ הַצֹּהַר הוּא רוֹאֶה אֶת הַגּוּפוֹת עַל פְּנֵי הַמַּיִם? וּכְשֶׁיָּרְדוּ הַמַּיִם — עָלָה הַכֹּל.

וְאָז נִפְתַּח הַמִּכְסֶה בַּפַּעַם הָרִאשׁוֹנָה, ״וְהִנֵּה חָרְבוּ״, וְהַגּוּף צָעַק לִסְגֹּר וְלַחְזֹר, אֶלָּא שֶׁבָּא הַקּוֹל ״צֵא מִן הַתֵּבָה״: לֹא שְׁאֵלָה, אֶלָּא צַו, לָצֵאת בִּכְפִיָּה אֶל מַה שֶּׁנִּשְׁאַר. וְהַלֵּב נִפְרָץ בְּהִתְקֵף הַחֲרָדָה הַגָּדוֹל מִכֻּלָּם, וְצוֹרֵחַ ״יָכֹלְתָּ לִמְנֹעַ, אֶת כָּל הַהֶרֶס הַזֶּה, אֶת כָּל הַגּוּפוֹת״, וְהוּא אֵינוֹ מַבְדִּיל בֵּין כִּשָּׁלוֹן לְאַשְׁמָה. ״וַיֵּצֵא נֹחַ״.

וּכְשֶׁבָּכָה, פַּעַם אַחַת בִּלְבַד, שָׁמַע ״רוֹעֶה שׁוֹטֶה, עַכְשָׁו?״ — דִּמְעָה אַחַת, וּתְשׁוּבָתָהּ ״הָיִיתָ צָרִיךְ״, וְכָךְ חוֹזְרִים הַבַּיְתָה. וּמֵאָז זֶה הֲכִי נֹחַ: לְהַסְבִּיר לָמָּה הוּא לֹא אַבְרָהָם, בִּזְמַן שֶׁהוּא עוֹד לֹא חָזַר לִהְיוֹת נֹחַ.

וּבְכָל זֹאת בָּנָה מִזְבֵּחַ, אֲבָנִים עַל אֲבָנִים, כְּדֵי שֶׁיִּהְיֶה לוֹ אֵיפֹה לַעֲמֹד — הָרִפּוּי הָרִאשׁוֹן. הוּא בִּקֵּשׁ חִבּוּק בְּלִי מִלִּים, וְהַחִבּוּק בָּא בְּמִלִּים: ״לֹא אֹסִף עוֹד״, לֹא עוֹד חֻרְבָּן כָּזֶה, לְעוֹלָם; וּבְאוֹת לְכָךְ קֶשֶׁת, שֶׁעָלֶיהָ כָּתַב בֶּן־נַחְמָן, וְנֶחָמָה בִּשְׁמוֹ, שֶׁהִיא קֶשֶׁת מִלְחָמָה: נֶשֶׁק מֻנָּח, הָפוּךְ, שֶׁאֵינוֹ מְכֻוָּן אֶל הָאָרֶץ.

״וַיָּחֶל״ — אַרְבָּעָה עוֹלָמוֹת בְּמִלָּה אַחַת: לְיַחֵל, לְהַתְחִיל, לְהִתְחַלֵּל, וְחָלָל. הוּא הִתְחִיל מִיָּד, וְהָעוֹלָם אָמַר ״יָפֶה, אַתָּה שׁוּב אַתָּה״, וְהוּא חִיֵּךְ בְּדִיּוּק כְּפִי שֶׁצִּפּוּ מִמֶּנּוּ. ״וַיִּטַּע כָּרֶם״, כִּי לִנְטֹעַ אַחֲרֵי הַחֻרְבָּן פֵּרוּשׁוֹ לְהָעֵז לְהַאֲמִין שֶׁלֶּעָתִיד יֵשׁ שָׁרָשִׁים.

אֶלָּא שֶׁאָז הִגִּיעַ הַשֶּׁקֶט, כְּמוֹ אוֹרֵחַ שֶׁלֹּא הוּזְמַן, וְהֵבִיא אִתּוֹ אֶת מַה שֶּׁהִתְעַכֵּב בַּדֶּרֶךְ: זִכָּרוֹן שֶׁאֵינוֹ יוֹדֵעַ שֶׁהוּא נִגְמַר, וְאֶצְלוֹ כָּל הַלַּיְלָה הוּא עַכְשָׁו; רֵיחַ אֶחָד, וְהוּא כְּבָר בִּפְנִים; דֶּלֶת שֶׁנִּטְרֶקֶת, וְהַגּוּף עַל הָרַגְלַיִם. וְהוּא נֶעְלַם מִדְּרָכִים שֶׁיֵּשׁ בָּהֶן קוֹלוֹת וּמֵאֲנָשִׁים, וְהָעוֹלָם אָמַר ״נוּ, כְּבָר עָבַר״, וְלֹא הֵבִין שֶׁמִּי שֶׁיָּצָא מִן הַתֵּבָה לֹא יָצָא מִמֶּנָּה.

וּבְעֶרֶב אֶחָד הֵרִים כּוֹס, ״וָיֶשְׁתְּ מִן הַיַּיִן״, וּבַתְּחִלָּה עָשָׂה לוֹ הַיַּיִן חֶסֶד — שָׁעָה שֶׁל לֵב שֶׁלֹּא דּוֹפֵק — וְאַחַר כָּךְ לָקַח אֶת הַכֹּל בַּחֲזָרָה, בַּלַּיְלָה, עִם רִבִּית, וְלָכֵן שָׁתָה עוֹד. כָּךְ נוֹפְלִים רַבִּים, לַיַּיִן וְלַכַּדּוּר וְלָעֲבוֹדָה, רַק לֹא לִרְאוֹת שׁוּב אֶת הַמַּרְאוֹת הָאֵלֶּה — וְיֵשׁ שֶׁלֹּא יִשְׁתּוּ אַף טִפָּה, וְהַכְּאֵב אוֹתוֹ כְּאֵב.

הוּא קָבַר בְּלִבּוֹ עוֹלָם שָׁלֵם, וּבִקֵּשׁ רַק דָּבָר אֶחָד: סִגְרוּ אֶת הַגּוֹלָל, וּתְנוּ לִי לְהִתְגּוֹלֵל בְּשֶׁקֶט; וְהַיַּיִן הָיָה הָאֶבֶן שֶׁמְּגַלְגְּלִים עַל הַקֶּבֶר שֶׁבַּלֵּב.

״וַיִּתְגַּל בְּתוֹךְ אׇהֳלֹה״ — לֹא אָדָם גִּלָּה אוֹתוֹ אֶלָּא הוּא נִגְלָה מִבִּפְנִים, וְהַכְּאֵב נִכְנַס הַבַּיְתָה, אֶל הַמִּטָּה וְאֶל הַיְּלָדִים שֶׁלֹּא מְבִינִים לָמָּה אַבָּא כָּאן וְאֵינוֹ כָּאן: גּוֹלֶה בְּתוֹךְ בֵּיתוֹ, שֶׁיָּצָא מִן הַתֵּבָה וְחֵלֶק מִמֶּנּוּ נִשְׁאַר בִּפְנִים. וְשׁוּב זֶה הֲכִי נֹחַ: לְסַפֵּר עַל הַיַּיִן, וְלֹא עַל הָעוֹלָם שֶׁקָּבַר בְּלִבּוֹ.

וְהָיָה מִי שֶׁ״וַיַּגֵּד בַּחוּץ״, וְדוֹרוֹת אַחַר כָּךְ עוֹד אָמְרוּ עָלָיו בֵּינוֹנִי — לֹא הִתְפַּלֵּל עַל דּוֹרוֹ, וּרְאוּ, בַּסּוֹף יָצָא וְשָׁתָה — וְאִישׁ לֹא שָׁאַל מָה רָאִיתָ, וְאִישׁ לֹא שָׁאַל אֵיךְ אַתָּה, כִּי כָּךְ קוֹרְאִים לְאָדָם בֵּינוֹנִי כְּשֶׁאֵין כֹּחַ לָדַעַת מָה הוּא נָשָׂא.

אֲבָל שְׁנַיִם לֹא הִסְבִּירוּ וְלֹא הִשְׁווּ וְלֹא צָחֲקוּ. שֵׁם, שֶׁהִתְחִיל בְּמִצְוָה, לֹא שָׁאַל מָה עָשָׂה, כִּי יָדַע מָה עָבַר עָלָיו, ו״וַיִּקַּח שֵׁם וָיֶפֶת אֶת הַשִּׂמְלָה״ וְהָלְכוּ אֲחוֹרַנִּית. לֹא רָאוּ אֶת עֶרְוַת אֲבִיהֶם, וְרָאוּ אֶת אֲבִיהֶם — אֶת הָאִישׁ, וְלֹא אֶת הַבֵּינוֹנִי.

״וַיִּיקֶץ נֹחַ מִיֵּינוֹ — וַיֵּדַע״, יָדַע שֶׁרָאוּ אוֹתוֹ וְכִסּוּ, וְאוּלַי אָז יָכוֹל הָיָה לְהוֹרִיד אֶת הַצִּפִּיָּה הָרִאשׁוֹנָה: מִי שֶׁנּוֹלַד כְּדֵי לְנַחֵם אֶת כֻּלָּם צָרִיךְ, אַחֲרֵי הַחֻרְבָּן, צִפִּיָּה אַחֶרֶת — לֹא לְהַצִּיל, לֹא לְתַקֵּן, אֶלָּא לָקוּם בַּבֹּקֶר וְלִנְשֹׁם וּלְהִשָּׁאֵר, כִּי רַק כְּשֶׁהַצִּפִּיָּה יוֹרֶדֶת אֶפְשָׁר לַחְזֹר לִשְׂפִיּוּת. הַמִּקְרָא אֵינוֹ מְסַפֵּר אֵיךְ נִרְפָּא, רַק שֶׁהַחַיִּים הִמְשִׁיכוּ מִמֶּנּוּ, כִּי לֹא כָּל מִי שֶׁיָּצָא מִן הַמַּבּוּל — הַמַּבּוּל יָצָא מִמֶּנּוּ.

וּמִנֹּחַ לָמַדְנוּ אֵיךְ נִצָּלִים מִן הַמַּבּוּל, וּמִשֵּׁם וָיֶפֶת — מָה עוֹשִׂים כְּשֶׁהַמַּבּוּל עוֹדֶנּוּ בְּתוֹכוֹ. יֵשׁ מִי שֶׁעָמַד בַּשּׁוּרָה וַעֲדַיִן עוֹמֵד, כִּי נֹחַ מִסְתּוֹבֵב בֵּינֵינוּ, נִרְאֶה נִינוֹחַ, ״כַּמָּה נוֹחַ לוֹ״, וּבֶאֱמֶת זֶה הֲכִי נוֹחַ — שֶׁלֹּא לִרְאוֹת. אֲבָל זֶה לֹא יִהְיֶה נוֹחַ, כִּי נִצְטָרֵךְ לִהְיוֹת שָׁם.

הוּא מְבַקֵּשׁ דָּבָר אֶחָד, שֶׁנִּזְכֹּר, לֹא אֶת הַמַּבּוּל אֶלָּא אוֹתוֹ, שֶׁנִּקְרָא לוֹ בִּשְׁמוֹ וְלֹא ״הַהוּא״, וְשֶׁנְּכַבֵּד אֶת מַה שֶּׁאֵינוֹ יַצִּיב בּוֹ. וְאִם יְבַקֵּשׁ שֶׁנִּסְגֹּר עָלָיו אֶת הַגּוֹלָל, לֹא נִסְגֹּר, אֶלָּא נֵשֵׁב לְיָדוֹ עַל הַקַּרְקַע בִּזְמַן שֶׁהוּא מִתְגּוֹלֵל, כִּי קָרְאוּ לוֹ נֹחַ כְּדֵי שֶׁיְּנַחֵם אוֹתָנוּ, וְהַפַּעַם אֲנַחְנוּ נְנַחֵם אוֹתוֹ.

אֲנַחְנוּ מִבְּנֵי שֵׁם, לֹא מִפְּנֵי שֶׁתָּמִיד הִצְלַחְנוּ לְכַסּוֹת, אֶלָּא מִפְּנֵי שֶׁעַל כָּתֵף אֶחָד מֵאֲבוֹתֵינוּ הֻנְּחָה שִׂמְלָה; וּבְנֵי שֵׁם נִדְרָשִׁים לִהְיוֹת שָׁם, וְלָתֵת שֵׁם, וְלִהְיוֹת בְּנֵי שֵׁם, לַעֲמֹד לְיַד הַדֶּלֶת וּלְהִשָּׁאֵר, וְלֹא רַק בַּשִּׁבְעָה. ״וַיִּזְכֹּר אֱלֹהִים אֶת נֹחַ״ — שֶׁנִּזְכֹּר גַּם אֲנַחְנוּ.
